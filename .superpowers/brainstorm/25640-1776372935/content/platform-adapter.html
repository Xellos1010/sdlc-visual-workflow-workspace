TYPESCRIPT INTERFACE
interface PlatformAdapter {
  auth: AuthAdapter
  db: DatabaseAdapter
  realtime: RealtimeAdapter
  storage: StorageAdapter
  events: EventAdapter
}

const platform = PlatformFactory
  .create(process.env.PLATFORM)
RUST TRAIT
trait PlatformAdapter:
  Send + Sync {
  async fn query(&self, ...) -> Result
  async fn emit_event(&self, ...) -> Result
  async fn upload(&self, ...) -> Result
}
